// 颜色定义映射表
const COLORS = {
  '黑': { digit: 0, multiplier: 1, tolerance: null, hex: '#000000' },
  '棕': { digit: 1, multiplier: 10, tolerance: 1, hex: '#8B4513' },
  '红': { digit: 2, multiplier: 100, tolerance: 2, hex: '#FF0000' },
  '橙': { digit: 3, multiplier: 1000, tolerance: null, hex: '#FFA500' },
  '黄': { digit: 4, multiplier: 10000, tolerance: null, hex: '#FFFF00' },
  '绿': { digit: 5, multiplier: 100000, tolerance: 0.5, hex: '#00FF00' },
  '蓝': { digit: 6, multiplier: 1000000, tolerance: 0.25, hex: '#0000FF' },
  '紫': { digit: 7, multiplier: 10000000, tolerance: 0.1, hex: '#800080' },
  '灰': { digit: 8, multiplier: 100000000, tolerance: 0.05, hex: '#808080' },
  '白': { digit: 9, multiplier: 1000000000, tolerance: null, hex: '#FFFFFF' },
  '金': { digit: null, multiplier: 0.1, tolerance: 5, hex: '#FFD700' },
  '银': { digit: null, multiplier: 0.01, tolerance: 10, hex: '#C0C0C0' }
};

/**
 * 将色环转换为电阻值
 * 
 * @param {string[]} colors - 色环颜色数组，按顺序排列（从左到右）
 * @param {number} bandCount - 色环数量，必须是 4 或 5
 * @returns {Object} 包含阻值和误差的对象
 *   - resistance {number}: 电阻值，单位欧姆（Ω）
 *   - tolerance {number}: 误差百分比（%）
 *   - unit {string}: 单位，固定为 'Ω'
 * @throws {Error} 当输入参数无效时抛出异常
 *   - colors 不是数组
 *   - bandCount 不是 4 或 5
 *   - colors 数组长度与 bandCount 不匹配
 *   - 包含无效的颜色名称
 *   - 四环电阻前两环使用金色或银色
 *   - 五环电阻前三环使用金色或银色
 *   - 误差环使用黑色、橙色、黄色或白色
 * @example
 * // 四环电阻示例
 * colorsToResistance(['棕', '黑', '红', '金'], 4)
 * // 返回: { resistance: 1000, tolerance: 5, unit: 'Ω' }
 * 
 * @example
 * // 五环电阻示例
 * colorsToResistance(['黄', '紫', '黑', '红', '棕'], 5)
 * // 返回: { resistance: 4700, tolerance: 1, unit: 'Ω' }
 * 
 * @example
 * // 小阻值电阻（使用金色倍率）
 * colorsToResistance(['棕', '黑', '银', '金'], 4)
 * // 返回: { resistance: 0.1, tolerance: 5, unit: 'Ω' }
 */
function colorsToResistance(colors, bandCount) {
  // 验证输入参数
  if (!Array.isArray(colors)) {
    throw new Error('colors 必须是数组');
  }
  if (bandCount !== 4 && bandCount !== 5) {
    throw new Error('bandCount 必须是 4 或 5');
  }
  if (colors.length !== bandCount) {
    throw new Error(`colors 数组长度必须是 ${bandCount}`);
  }
  
  // 验证每个颜色是否有效
  for (let i = 0; i < colors.length; i++) {
    if (!COLORS[colors[i]]) {
      throw new Error(`无效的颜色: ${colors[i]}`);
    }
  }
  
  let resistance;
  let tolerance;
  
  if (bandCount === 4) {
    // 四环电阻：2位有效数字 + 倍率 + 误差
    const digit1 = COLORS[colors[0]].digit;
    const digit2 = COLORS[colors[1]].digit;
    const multiplier = COLORS[colors[2]].multiplier;
    tolerance = COLORS[colors[3]].tolerance;
    
    // 验证有效数字环不能是金色或银色
    if (digit1 === null || digit2 === null) {
      throw new Error('四环电阻的前两环不能是金色或银色');
    }
    
    // 计算阻值
    resistance = (digit1 * 10 + digit2) * multiplier;
  } else {
    // 五环电阻：3位有效数字 + 倍率 + 误差
    const digit1 = COLORS[colors[0]].digit;
    const digit2 = COLORS[colors[1]].digit;
    const digit3 = COLORS[colors[2]].digit;
    const multiplier = COLORS[colors[3]].multiplier;
    tolerance = COLORS[colors[4]].tolerance;
    
    // 验证有效数字环不能是金色或银色
    if (digit1 === null || digit2 === null || digit3 === null) {
      throw new Error('五环电阻的前三环不能是金色或银色');
    }
    
    // 计算阻值
    resistance = (digit1 * 100 + digit2 * 10 + digit3) * multiplier;
  }
  
  // 验证误差环
  if (tolerance === null) {
    throw new Error('误差环不能是黑色、橙色、黄色或白色');
  }
  
  return {
    resistance: resistance,
    tolerance: tolerance,
    unit: 'Ω'
  };
}

/**
 * 将电阻值转换为色环
 * 
 * @param {number} resistance - 电阻值，单位欧姆（Ω），必须 >= 0
 * @param {number} tolerance - 误差百分比，必须是以下值之一：0.05, 0.1, 0.25, 0.5, 1, 2, 5, 10
 * @param {number} bandCount - 色环数量，必须是 4 或 5
 * @returns {string[]} 色环颜色数组，按顺序排列（从左到右）
 * @throws {Error} 当输入参数无效时抛出异常
 *   - resistance 不是数字或为 NaN
 *   - resistance 为负数
 *   - tolerance 不是数字或为 NaN
 *   - tolerance 不在有效值列表中
 *   - bandCount 不是 4 或 5
 * @example
 * // 四环电阻示例
 * resistanceToColors(1000, 5, 4)
 * // 返回: ['棕', '黑', '红', '金']
 * 
 * @example
 * // 五环电阻示例
 * resistanceToColors(4700, 1, 5)
 * // 返回: ['黄', '紫', '黑', '红', '棕']
 * 
 * @example
 * // 0 欧姆电阻
 * resistanceToColors(0, 5, 4)
 * // 返回: ['黑', '黑', '黑', '金']
 */
function resistanceToColors(resistance, tolerance, bandCount) {
  // 验证输入参数
  if (typeof resistance !== 'number' || isNaN(resistance)) {
    throw new Error('resistance 必须是数字');
  }
  if (resistance < 0) {
    throw new Error('resistance 不能为负数');
  }
  if (typeof tolerance !== 'number' || isNaN(tolerance)) {
    throw new Error('tolerance 必须是数字');
  }
  if (bandCount !== 4 && bandCount !== 5) {
    throw new Error('bandCount 必须是 4 或 5');
  }
  
  // 验证误差值
  const validTolerances = [0.05, 0.1, 0.25, 0.5, 1, 2, 5, 10];
  if (!validTolerances.includes(tolerance)) {
    throw new Error(`无效的误差值: ${tolerance}，必须是 ${validTolerances.join(', ')} 之一`);
  }
  
  // 找到误差颜色
  let toleranceColor = null;
  for (const [color, data] of Object.entries(COLORS)) {
    if (data.tolerance === tolerance) {
      toleranceColor = color;
      break;
    }
  }
  
  if (!toleranceColor) {
    throw new Error(`找不到误差 ${tolerance}% 对应的颜色`);
  }
  
  // 处理特殊情况：0 欧姆
  if (resistance === 0) {
    if (bandCount === 4) {
      return ['黑', '黑', '黑', toleranceColor];
    } else {
      return ['黑', '黑', '黑', '黑', toleranceColor];
    }
  }
  
  // 计算有效数字和倍率
  let digits;
  let multiplierValue;
  let multiplierColor;
  
  if (bandCount === 4) {
    // 四环：2位有效数字（10-99）
    let tempResistance = resistance;
    multiplierValue = 1;
    
    // 调整到 10-99 范围
    while (tempResistance >= 100) {
      tempResistance /= 10;
      multiplierValue *= 10;
    }
    while (tempResistance < 10 && multiplierValue > 0.01) {
      tempResistance *= 10;
      multiplierValue /= 10;
    }
    
    // 四舍五入到整数
    digits = Math.round(tempResistance);
    
    // 如果四舍五入后超出范围，调整倍率
    if (digits >= 100) {
      digits = Math.round(digits / 10);
      multiplierValue *= 10;
    } else if (digits < 10) {
      digits = Math.round(digits * 10);
      multiplierValue /= 10;
    }
    
    // 找到倍率颜色
    multiplierColor = findMultiplierColor(multiplierValue);
    
    const digit1 = Math.floor(digits / 10);
    const digit2 = digits % 10;
    
    return [
      findDigitColor(digit1),
      findDigitColor(digit2),
      multiplierColor,
      toleranceColor
    ];
  } else {
    // 五环：3位有效数字（100-999）
    let tempResistance = resistance;
    multiplierValue = 1;
    
    // 调整到 100-999 范围
    while (tempResistance >= 1000) {
      tempResistance /= 10;
      multiplierValue *= 10;
    }
    while (tempResistance < 100 && multiplierValue > 0.01) {
      tempResistance *= 10;
      multiplierValue /= 10;
    }
    
    // 四舍五入到整数
    digits = Math.round(tempResistance);
    
    // 如果四舍五入后超出范围，调整倍率
    if (digits >= 1000) {
      digits = Math.round(digits / 10);
      multiplierValue *= 10;
    } else if (digits < 100) {
      digits = Math.round(digits * 10);
      multiplierValue /= 10;
    }
    
    // 找到倍率颜色
    multiplierColor = findMultiplierColor(multiplierValue);
    
    const digit1 = Math.floor(digits / 100);
    const digit2 = Math.floor((digits % 100) / 10);
    const digit3 = digits % 10;
    
    return [
      findDigitColor(digit1),
      findDigitColor(digit2),
      findDigitColor(digit3),
      multiplierColor,
      toleranceColor
    ];
  }
}

/**
 * 根据数字找到对应的颜色
 * 
 * @param {number} digit - 数字，范围 0-9
 * @returns {string} 对应的颜色名称
 * @throws {Error} 当数字超出 0-9 范围时抛出异常
 * @example
 * findDigitColor(0) // 返回: '黑'
 * findDigitColor(5) // 返回: '绿'
 * findDigitColor(9) // 返回: '白'
 */
function findDigitColor(digit) {
  for (const [color, data] of Object.entries(COLORS)) {
    if (data.digit === digit) {
      return color;
    }
  }
  throw new Error(`找不到数字 ${digit} 对应的颜色`);
}

/**
 * 根据倍率找到对应的颜色
 * 
 * @param {number} multiplier - 倍率值
 * @returns {string} 对应的颜色名称
 * @throws {Error} 当找不到匹配的倍率颜色时抛出异常
 * @example
 * findMultiplierColor(1) // 返回: '黑'
 * findMultiplierColor(1000) // 返回: '橙'
 * findMultiplierColor(0.1) // 返回: '金'
 */
function findMultiplierColor(multiplier) {
  // 使用 epsilon 处理浮点误差
  const epsilon = 1e-10;
  
  for (const [color, data] of Object.entries(COLORS)) {
    if (Math.abs(data.multiplier - multiplier) < epsilon) {
      return color;
    }
  }
  
  // 如果找不到精确匹配，找最接近的
  let closestColor = null;
  let minDiff = Infinity;
  
  for (const [color, data] of Object.entries(COLORS)) {
    const diff = Math.abs(data.multiplier - multiplier);
    if (diff < minDiff) {
      minDiff = diff;
      closestColor = color;
    }
  }
  
  return closestColor;
}

/**
 * 格式化电阻值显示
 * 
 * @param {number} value - 电阻值，单位欧姆（Ω）
 * @returns {string} 格式化后的字符串，自动选择合适的单位
 * @example
 * formatResistance(1000) // 返回: "1kΩ"
 * formatResistance(0.001) // 返回: "1mΩ"
 * formatResistance(1000000) // 返回: "1MΩ"
 * formatResistance(4700) // 返回: "4.7kΩ"
 */
function formatResistance(value) {
  if (value === 0) return '0Ω';
  
  const units = [
    { threshold: 1e9, suffix: 'GΩ' },
    { threshold: 1e6, suffix: 'MΩ' },
    { threshold: 1e3, suffix: 'kΩ' },
    { threshold: 1, suffix: 'Ω' },
    { threshold: 1e-3, suffix: 'mΩ' },
    { threshold: 1e-6, suffix: 'μΩ' }
  ];
  
  for (const unit of units) {
    if (Math.abs(value) >= unit.threshold) {
      const scaled = value / unit.threshold;
      // 保留适当的小数位数，去除末尾的 0
      const formatted = scaled % 1 === 0 
        ? scaled.toString() 
        : scaled.toFixed(2).replace(/\.?0+$/, '');
      return formatted + unit.suffix;
    }
  }
  
  return value.toString() + 'Ω';
}

一、工具介绍

色环电阻计算器是一个用于四环和五环电阻色环与阻值双向换算的工具。它解决了电子工程师、电子爱好者和学生在实际工作中遇到的两个核心问题：

  1. 色环识别：当拿到一个实物电阻时，通过读取色环颜色快速计算出阻值和误差范围
  2. 色环标注：当需要某个特定阻值的电阻时，快速确定应该选择什么色环组合的电阻

本工具严格依据国际电工委员会标准 IEC 60062（《Resistor and capacitor colour codes and marking codes》）实现，确保计算结果的准确性和可靠性。

二、参数详细说明

2.1 色环颜色参数（colors）

色环颜色是一个字符串数组，按从左到右的顺序排列。每个元素必须是以下 12 种颜色之一：

颜色 有效数字 倍率 误差（%） 说明
黑 0 ×1 - 不能用于误差环
棕 1 ×10 ±1% 常用误差色
红 2 ×100 ±2% 常用误差色
橙 3 ×1k - 不能用于误差环
黄 4 ×10k - 不能用于误差环
绿 5 ×100k ±0.5% 精密电阻常用
蓝 6 ×1M ±0.25% 精密电阻常用
紫 7 ×10M ±0.1% 高精度电阻
灰 8 ×100M ±0.05% 超高精度电阻
白 9 ×1G - 不能用于误差环
金 - ×0.1 ±5% 最常用误差色
银 - ×0.01 ±10% 低精度电阻

2.2 色环数量参数（bandCount）

色环数量必须是 4 或 5：

  • 四环电阻：2位有效数字 + 倍率 + 误差
  • 五环电阻：3位有效数字 + 倍率 + 误差（精度更高）

2.3 电阻值参数（resistance）

电阻值以欧姆（Ω）为单位，必须是非负数。取值范围：

  • 四环电阻：0.1Ω ~ 99GΩ
  • 五环电阻：0.01Ω ~ 999GΩ

2.4 误差参数（tolerance）

误差百分比必须是以下值之一：0.05, 0.1, 0.25, 0.5, 1, 2, 5, 10。这些值对应不同的颜色，详见上表。

三、实际使用场景

场景 1：识别实物电阻的阻值

情况：你从电阻盒里拿出一个电阻，色环从左到右依次是：棕、黑、红、金

输入：
colors = ['棕', '黑', '红', '金']
bandCount = 4
计算过程：
第1环（棕）= 1
第2环（黑）= 0
第3环（红）= ×100
第4环（金）= ±5%
阻值 = (1×10 + 0) × 100 = 1000Ω = 1kΩ
结果：1kΩ ±5%
实际阻值范围：950Ω ~ 1050Ω

场景 2：为电路设计选择电阻

情况：设计一个 LED 限流电路，需要 4.7kΩ 的电阻，误差要求 ±1%

输入：
resistance = 4700
tolerance = 1
bandCount = 5（选择五环以获得更高精度）
计算过程：
有效数字 = 470
倍率 = ×10
误差 = ±1%（棕色）
结果：色环为 黄、紫、黑、棕、棕
购买建议：购买五环电阻，色环顺序为黄紫黑棕棕

场景 3：识别小阻值电阻

情况：在电流检测电路中看到一个电阻，色环为：棕、黑、银、金

输入：
colors = ['棕', '黑', '银', '金']
bandCount = 4
计算过程：
第1环（棕）= 1
第2环（黑）= 0
第3环（银）= ×0.01
第4环（金）= ±5%
阻值 = (1×10 + 0) × 0.01 = 0.1Ω
结果：0.1Ω ±5%
应用场景：常用于电流检测、采样电阻

场景 4：高精度电路设计

情况：设计精密测量电路，需要 680Ω 电阻，误差要求 ±0.1%

输入：
resistance = 680
tolerance = 0.1
bandCount = 5
计算过程：
有效数字 = 680
倍率 = ×1（黑色）
误差 = ±0.1%（紫色）
结果：色环为 蓝、灰、黑、黑、紫
注意事项：这种高精度电阻价格较高，需确认是否必要

场景 5：大阻值电阻识别

情况：在高压电路中看到一个电阻，色环为：橙、橙、绿、棕

输入：
colors = ['橙', '橙', '绿', '棕']
bandCount = 4
计算过程：
第1环（橙）= 3
第2环（橙）= 3
第3环（绿）= ×100k
第4环（棕）= ±1%
阻值 = (3×10 + 3) × 100000 = 3300000Ω = 3.3MΩ
结果：3.3MΩ ±1%
应用场景：常用于高压分压、泄放电阻

四、常见错误与排查

错误 1：色环顺序读反

现象：计算出的阻值明显不合理（如应该几 kΩ 却算出几 MΩ）

原因：色环电阻的误差环（金、银、棕、红等）通常与其他环间距较大，或者靠近电阻一端

解决：找到间距较大的那一环作为误差环，从另一端开始读取

错误 2：颜色识别错误

现象：计算结果与实际测量值不符

原因：色环褪色、光线不足、色盲等原因导致颜色识别错误

解决：

  • 在充足光线下识别
  • 使用万用表测量验证
  • 注意区分相近颜色（如红与橙、蓝与紫）

错误 3：四环和五环混淆

现象：输入参数错误导致异常

原因：没有正确识别电阻是四环还是五环

解决：数清楚色环数量，四环电阻有 4 道色环，五环电阻有 5 道色环

错误 4：误差值不在有效范围

现象：程序抛出"无效的误差值"异常

原因：输入的误差值不是标准值（如输入 3%）

解决：只能使用标准误差值：0.05, 0.1, 0.25, 0.5, 1, 2, 5, 10

五、数据来源与标准依据

5.1 国际标准

本工具严格依据以下国际标准实现：

  • IEC 60062：Resistor and capacitor colour codes and marking codes（电阻器和电容器色标和标注代码）
  • IEC 60063：Preferred number series for resistors and capacitors（电阻器和电容器的优先数系）

5.2 国家标准

对应的中国国家标准：

  • GB/T 2481-1998：电阻器和电容器标志代码（等同采用 IEC 60062:1993）
注意：如需查阅标准原文，建议通过国家标准全文公开系统或购买正版标准文件。本工具的实现基于公开的技术资料和行业通用规范。

5.3 E 系列标准阻值

实际生产的电阻遵循 E 系列标准阻值：

  • E12 系列：12 个基本值，误差 ±10%（如 1.0, 1.2, 1.5, 1.8, 2.2, 2.7, 3.3, 3.9, 4.7, 5.6, 6.8, 8.2）
  • E24 系列：24 个基本值，误差 ±5%（在 E12 基础上增加 1.1, 1.3, 1.6, 2.0, 2.4, 3.0, 3.6, 4.3, 5.1, 6.2, 7.5, 9.1）
  • E96 系列：96 个基本值，误差 ±1%（更精细的阻值）

六、局限性说明

重要提示：本工具有以下局限性，使用时请注意：

6.1 不支持的电阻类型

  • 六环电阻：本工具不支持六环电阻（包含温度系数环）
  • 贴片电阻：不支持贴片电阻的数字标注法（如 103、472 等）
  • 特殊标注：不支持某些厂家的特殊标注方式

6.2 精度限制

  • 浮点精度：由于 JavaScript 使用 IEEE 754 双精度浮点数，极小或极大阻值可能存在微小误差（通常在 10^-10 级别）
  • 四舍五入：阻值转色环时会对有效数字进行四舍五入，可能与实际标准阻值有偏差

6.3 功能限制

  • 温度系数：不支持温度系数（TCR）的计算和标注
  • 功率 rating：不支持电阻功率的计算
  • 非标阻值：输入的阻值如果不是标准 E 系列值，转换后的色环可能对应最接近的标准值

6.4 实际应用注意

  • 实物识别：本工具不能替代万用表测量，对于关键电路建议实际测量验证
  • 色环褪色：老旧电阻的色环可能褪色，导致识别困难
  • 温度影响：电阻值会随温度变化，本工具计算的是标称值（25°C 时的值）
建议：对于精密电路设计，建议结合本工具计算结果和实际测量值，并考虑电阻的温度系数、功率裕量等因素。
