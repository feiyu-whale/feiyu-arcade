标准直齿圆柱齿轮计算工具

依据 GB/T 1356-2001 / ISO 53:1998 · 标准基本齿条齿廓 · 压力角 α=20°

JS 源码（纯函数实现，无外部依赖）

以下代码可直接在浏览器控制台运行。所有函数均为纯函数，输入合法返回数值，输入非法抛出 Error。

/**
 * 标准直齿圆柱齿轮计算库
 * 依据：GB/T 1356-2001《通用机械和重型机械用圆柱齿轮 标准基本齿条齿廓》
 *       (等同采用 ISO 53:1998)
 * 标准压力角 α = 20°，齿顶高系数 ha* = 1，顶隙系数 c* = 0.25
 * 适用条件：标准齿轮（变位系数 x = 0），无侧隙啮合
 */

(function (global) {
  'use strict';

  // ==================== 常量 ====================
  /** 标准压力角（度） */
  const STD_PRESSURE_ANGLE_DEG = 20;
  /** 标准压力角（弧度） */
  const STD_PRESSURE_ANGLE_RAD = STD_PRESSURE_ANGLE_DEG * Math.PI / 180;
  /** 标准齿顶高系数 ha* */
  const STD_HA_STAR = 1;
  /** 标准顶隙系数 c* */
  const STD_C_STAR = 0.25;
  /** 避免根切的最小齿数（α=20°，ha*=1）：z_min = 2*ha*/sin²α ≈ 17.097 */
  const MIN_TEETH_NO_UNDERCUT = 17;
  /** 模数下限（mm），参考 GB/T 1357 标准模数系列 */
  const MODULE_MIN = 0.05;
  /** 模数上限（mm），工程常用上限 */
  const MODULE_MAX = 100;
  /** 齿数下限 */
  const TEETH_MIN = 1;
  /** 齿数上限（工程实际限制，工具不做硬性限制，仅做合理性提示） */
  const TEETH_SOFT_MAX = 10000;

  // ==================== 内部工具 ====================
  /**
   * 校验数值是否为有限数
   * @param {*} v - 待校验值
   * @param {string} name - 参数名（用于错误信息）
   * @throws {Error} 非有限数时抛出
   */
  function assertFinite(v, name) {
    if (typeof v !== 'number' || !Number.isFinite(v)) {
      throw new Error(`参数 ${name} 必须为有限数值，当前值: ${v}`);
    }
  }

  /**
   * 校验齿数（必须为正整数）
   * @param {*} z - 齿数
   * @throws {Error} 非法齿数时抛出
   */
  function assertTeeth(z) {
    assertFinite(z, 'z');
    if (!Number.isInteger(z)) {
      throw new Error(`齿数 z 必须为整数，当前值: ${z}`);
    }
    if (z < TEETH_MIN) {
      throw new Error(`齿数 z 必须 >= ${TEETH_MIN}，当前值: ${z}`);
    }
  }

  /**
   * 校验模数（必须为正数且在合理范围）
   * @param {*} m - 模数（mm）
   * @throws {Error} 非法模数时抛出
   */
  function assertModule(m) {
    assertFinite(m, 'm');
    if (m <= 0) {
      throw new Error(`模数 m 必须 > 0，当前值: ${m}`);
    }
    if (m < MODULE_MIN || m > MODULE_MAX) {
      throw new Error(`模数 m 超出合理范围 [${MODULE_MIN}, ${MODULE_MAX}] mm，当前值: ${m}`);
    }
  }

  // ==================== 核心计算函数 ====================

  /**
   * 计算分度圆直径
   * 公式：d = m × z
   *
   * @param {number} m - 模数（mm），必须 > 0，推荐范围 [0.05, 100]
   * @param {number} z - 齿数，必须为正整数，推荐 >= 17（避免根切）
   * @returns {number} 分度圆直径 d（mm）
   * @throws {Error} 参数非法时抛出
   *
   * @example
   * calcPitchDiameter(2, 20);   // => 40
   * calcPitchDiameter(1.5, 30); // => 45
   * calcPitchDiameter(4, 25);   // => 100
   */
  function calcPitchDiameter(m, z) {
    assertModule(m);
    assertTeeth(z);
    return m * z;
  }

  /**
   * 计算齿顶圆直径
   * 公式：da = m × (z + 2 × ha*) = m × (z + 2)  （标准齿顶高系数 ha*=1）
   *
   * @param {number} m - 模数（mm），必须 > 0
   * @param {number} z - 齿数，必须为正整数
   * @returns {number} 齿顶圆直径 da（mm）
   * @throws {Error} 参数非法时抛出
   *
   * @example
   * calcAddendumDiameter(2, 20); // => 44
   * calcAddendumDiameter(1, 30); // => 32
   */
  function calcAddendumDiameter(m, z) {
    assertModule(m);
    assertTeeth(z);
    return m * (z + 2 * STD_HA_STAR);
  }

  /**
   * 计算齿根圆直径
   * 公式：df = m × (z - 2 × ha* - 2 × c*) = m × (z - 2.5)
   *        （ha*=1, c*=0.25）
   *
   * @param {number} m - 模数（mm），必须 > 0
   * @param {number} z - 齿数，必须为正整数；注意 z < 3 时 df 为负（无物理意义）
   * @returns {number} 齿根圆直径 df（mm）
   * @throws {Error} 参数非法时抛出
   *
   * @example
   * calcDedendumDiameter(2, 20); // => 35
   * calcDedendumDiameter(4, 25); // => 90
   */
  function calcDedendumDiameter(m, z) {
    assertModule(m);
    assertTeeth(z);
    return m * (z - 2 * STD_HA_STAR - 2 * STD_C_STAR);
  }

  /**
   * 计算标准中心距（两标准齿轮无侧隙啮合）
   * 公式：a = (d1 + d2) / 2 = m × (z1 + z2) / 2
   *
   * @param {number} m  - 模数（mm），必须 > 0，两齿轮模数必须相同
   * @param {number} z1 - 小齿轮齿数，正整数
   * @param {number} z2 - 大齿轮齿数，正整数
   * @returns {number} 标准中心距 a（mm）
   * @throws {Error} 参数非法时抛出
   *
   * @example
   * calcCenterDistance(2, 20, 40); // => 60
   * calcCenterDistance(3, 25, 75); // => 150
   */
  function calcCenterDistance(m, z1, z2) {
    assertModule(m);
    assertTeeth(z1);
    assertTeeth(z2);
    return m * (z1 + z2) / 2;
  }

  /**
   * 计算齿顶高
   * 公式：ha = ha* × m = m
   *
   * @param {number} m - 模数（mm）
   * @returns {number} 齿顶高 ha（mm）
   * @throws {Error} 参数非法时抛出
   *
   * @example
   * calcAddendumHeight(2); // => 2
   */
  function calcAddendumHeight(m) {
    assertModule(m);
    return STD_HA_STAR * m;
  }

  /**
   * 计算齿根高
   * 公式：hf = (ha* + c*) × m = 1.25 × m
   *
   * @param {number} m - 模数（mm）
   * @returns {number} 齿根高 hf（mm）
   * @throws {Error} 参数非法时抛出
   *
   * @example
   * calcDedendumHeight(4); // => 5
   */
  function calcDedendumHeight(m) {
    assertModule(m);
    return (STD_HA_STAR + STD_C_STAR) * m;
  }

  /**
   * 计算全齿高
   * 公式：h = ha + hf = (2 × ha* + c*) × m = 2.25 × m
   *
   * @param {number} m - 模数（mm）
   * @returns {number} 全齿高 h（mm）
   * @throws {Error} 参数非法时抛出
   *
   * @example
   * calcWholeHeight(2); // => 4.5
   */
  function calcWholeHeight(m) {
    assertModule(m);
    return (2 * STD_HA_STAR + STD_C_STAR) * m;
  }

  /**
   * 计算基圆直径
   * 公式：db = d × cos(α) = m × z × cos(20°)
   *
   * @param {number} m - 模数（mm）
   * @param {number} z - 齿数，正整数
   * @returns {number} 基圆直径 db（mm）
   * @throws {Error} 参数非法时抛出
   *
   * @example
   * calcBaseDiameter(2, 20); // => 37.5877...
   */
  function calcBaseDiameter(m, z) {
    assertModule(m);
    assertTeeth(z);
    return m * z * Math.cos(STD_PRESSURE_ANGLE_RAD);
  }

  /**
   * 计算齿距（周节）
   * 公式：p = π × m
   *
   * @param {number} m - 模数（mm）
   * @returns {number} 齿距 p（mm）
   * @throws {Error} 参数非法时抛出
   *
   * @example
   * calcCircularPitch(2); // => 6.2831...
   */
  function calcCircularPitch(m) {
    assertModule(m);
    return Math.PI * m;
  }

  /**
   * 计算单个齿轮的全部基本参数
   *
   * @param {number} m - 模数（mm）
   * @param {number} z - 齿数，正整数
   * @returns {{ m:number, z:number, d:number, da:number, df:number, db:number,
   *             ha:number, hf:number, h:number, p:number,
   *             alpha:number, haStar:number, cStar:number,
   *             hasUndercutRisk:boolean }} 计算结果对象
   * @throws {Error} 参数非法时抛出
   *
   * @example
   * calcAllParameters(2, 20);
   * // => { m:2, z:20, d:40, da:44, df:35, db:37.587..., ha:2, hf:2.5, ... }
   */
  function calcAllParameters(m, z) {
    assertModule(m);
    assertTeeth(z);
    return {
      m: m,
      z: z,
      d: calcPitchDiameter(m, z),
      da: calcAddendumDiameter(m, z),
      df: calcDedendumDiameter(m, z),
      db: calcBaseDiameter(m, z),
      ha: calcAddendumHeight(m),
      hf: calcDedendumHeight(m),
      h: calcWholeHeight(m),
      p: calcCircularPitch(m),
      alpha: STD_PRESSURE_ANGLE_DEG,
      haStar: STD_HA_STAR,
      cStar: STD_C_STAR,
      hasUndercutRisk: z < MIN_TEETH_NO_UNDERCUT
    };
  }

  /**
   * 计算一对标准齿轮副的全部参数
   *
   * @param {number} m  - 模数（mm），两齿轮必须相同
   * @param {number} z1 - 小齿轮齿数
   * @param {number} z2 - 大齿轮齿数
   * @returns {{ m:number, z1:number, z2:number,
   *             d1:number, da1:number, df1:number, db1:number,
   *             d2:number, da2:number, df2:number, db2:number,
   *             a:number, i:number, ha:number, hf:number, h:number, p:number,
   *             alpha:number, gear1HasUndercutRisk:boolean, gear2HasUndercutRisk:boolean }}
   * @throws {Error} 参数非法时抛出
   *
   * @example
   * calcPairParameters(2, 20, 40);
   * // => { m:2, z1:20, z2:40, a:60, i:2, d1:40, d2:80, ... }
   */
  function calcPairParameters(m, z1, z2) {
    assertModule(m);
    assertTeeth(z1);
    assertTeeth(z2);
    const g1 = calcAllParameters(m, z1);
    const g2 = calcAllParameters(m, z2);
    return {
      m: m,
      z1: z1,
      z2: z2,
      d1: g1.d, da1: g1.da, df1: g1.df, db1: g1.db,
      d2: g2.d, da2: g2.da, df2: g2.df, db2: g2.db,
      a: calcCenterDistance(m, z1, z2),
      i: z2 / z1,
      ha: g1.ha,
      hf: g1.hf,
      h: g1.h,
      p: g1.p,
      alpha: STD_PRESSURE_ANGLE_DEG,
      gear1HasUndercutRisk: g1.hasUndercutRisk,
      gear2HasUndercutRisk: g2.hasUndercutRisk
    };
  }

  /**
   * 判断给定齿数是否存在根切风险（标准齿轮 α=20°）
   *
   * @param {number} z - 齿数，正整数
   * @returns {boolean} z < 17 时返回 true
   * @throws {Error} 参数非法时抛出
   *
   * @example
   * hasUndercutRisk(15); // => true
   * hasUndercutRisk(20); // => false
   */
  function hasUndercutRisk(z) {
    assertTeeth(z);
    return z < MIN_TEETH_NO_UNDERCUT;
  }

  // ==================== 导出 ====================
  const Gear = {
    // 常量
    STD_PRESSURE_ANGLE_DEG: STD_PRESSURE_ANGLE_DEG,
    STD_HA_STAR: STD_HA_STAR,
    STD_C_STAR: STD_C_STAR,
    MIN_TEETH_NO_UNDERCUT: MIN_TEETH_NO_UNDERCUT,
    MODULE_MIN: MODULE_MIN,
    MODULE_MAX: MODULE_MAX,
    // 核心函数
    calcPitchDiameter: calcPitchDiameter,
    calcAddendumDiameter: calcAddendumDiameter,
    calcDedendumDiameter: calcDedendumDiameter,
    calcCenterDistance: calcCenterDistance,
    calcAddendumHeight: calcAddendumHeight,
    calcDedendumHeight: calcDedendumHeight,
    calcWholeHeight: calcWholeHeight,
    calcBaseDiameter: calcBaseDiameter,
    calcCircularPitch: calcCircularPitch,
    calcAllParameters: calcAllParameters,
    calcPairParameters: calcPairParameters,
    hasUndercutRisk: hasUndercutRisk
  };

  // 兼容浏览器与 Node
  if (typeof module !== 'undefined' && module.exports) {
    module.exports = Gear;
  } else {
    global.Gear = Gear;
  }
})(typeof window !== 'undefined' ? window : this);

使用说明书

一、工具简介

本工具用于计算标准直齿圆柱齿轮（即变位系数 x=0、压力角 α=20°、齿顶高系数 ha*=1、顶隙系数 c*=0.25 的渐开线圆柱齿轮）的基本几何参数。输入模数 m 和齿数 z，即可得到分度圆直径、齿顶圆直径、齿根圆直径、基圆直径、齿顶高、齿根高、全齿高、齿距等关键尺寸；输入一对齿轮的模数和齿数，还可得到标准中心距和传动比。

该工具解决的核心问题是：在机械设计中，当工程师确定了模数和齿数后，快速、准确地得到齿轮副的全部基本几何尺寸，避免手算出错，为后续绘图、加工、校核提供数据基础。

二、标准依据

  • GB/T 1356-2001《通用机械和重型机械用圆柱齿轮 标准基本齿条齿廓》（等同采用 ISO 53:1998）——规定标准齿条齿廓的基本参数：α=20°、ha*=1、c*=0.25。
  • GB/T 1357-2008《通用机械和重型机械用圆柱齿轮 模数》——规定标准模数系列（第一系列：0.05, 0.06, 0.08, 0.10, 0.12, 0.16, 0.20, 0.25, 0.3, 0.4, 0.5, 0.6, 0.8, 1, 1.25, 1.5, 2, 2.5, 3, 4, 5, 6, 8, 10, 12, 16, 20, 25, 32, 40, 50）。
  • GB/T 10095 系列——圆柱齿轮精度制（本工具不涉及精度等级计算）。
注：本工具仅实现标准齿轮（x=0）的基本几何计算。变位齿轮、斜齿轮、锥齿轮、蜗轮蜗杆等不在本工具范围内。

三、参数详细说明

参数含义单位取值范围说明
m模数mm(0, 100]，推荐 GB/T 1357 标准值齿轮最基本的参数，决定齿的大小。必须为正数。超出 [0.05, 100] 工具会抛出异常。
z齿数—正整数，≥1必须为整数。标准齿轮 α=20° 时，z<17 会出现根切，工具会返回 hasUndercutRisk=true 提示。
z1, z2小/大齿轮齿数—正整数用于齿轮副计算，两齿轮模数必须相同。
d分度圆直径mm—齿轮设计的基准圆，d = m·z。
da齿顶圆直径mm—齿顶所在的圆，da = m·(z+2)。
df齿根圆直径mm—齿槽底所在的圆，df = m·(z−2.5)。z<3 时 df 为负，无物理意义。
db基圆直径mm—渐开线发生的圆，db = d·cos(20°)。
ha齿顶高mm—分度圆到齿顶圆的径向距离，ha = m。
hf齿根高mm—分度圆到齿根圆的径向距离，hf = 1.25m。
h全齿高mm—齿顶圆到齿根圆的径向距离，h = 2.25m。
p齿距（周节）mm—分度圆上相邻两齿同侧齿廓间的弧长，p = π·m。
a标准中心距mm—两标准齿轮无侧隙啮合时的中心距，a = m·(z1+z2)/2。
i传动比——i = z2/z1。

四、核心公式

d = m × z
da = m × (z + 2)
df = m × (z − 2.5)
db = m × z × cos(20°)
ha = m ；hf = 1.25m ；h = 2.25m
p = π × m
a = m × (z1 + z2) / 2

五、实际使用场景（5 个）

场景 1：普通减速器小齿轮设计

需求：某带式输送机减速器，输入轴转速 1440 r/min，输出轴 360 r/min，传动比 i=4，初步选定模数 m=2.5 mm，小齿轮齿数 z1=25。

输入：m=2.5，z1=25，z2=100（i=4）

调用：Gear.calcPairParameters(2.5, 25, 100)

结果：

  • 小齿轮：d1=62.5 mm，da1=67.5 mm，df1=56.25 mm
  • 大齿轮：d2=250 mm，da2=255 mm，df2=243.75 mm
  • 标准中心距 a = 156.25 mm
  • 传动比 i = 4
  • 两齿轮均无根切风险（z1=25 > 17）

场景 2：仪器仪表微型齿轮

需求：某精密仪器中一对传动齿轮，空间受限，选用模数 m=0.5 mm，齿数 z1=20，z2=60。

输入：m=0.5，z1=20，z2=60

调用：Gear.calcPairParameters(0.5, 20, 60)

结果：

  • 小齿轮：d1=10 mm，da1=11 mm，df1=8.75 mm
  • 大齿轮：d2=30 mm，da2=31 mm，df2=28.75 mm
  • 标准中心距 a = 20 mm
  • 传动比 i = 3

场景 3：重型机械大模数齿轮

需求：矿山破碎机传动，大模数齿轮，m=10 mm，z1=20，z2=80。

输入：m=10，z1=20，z2=80

调用：Gear.calcPairParameters(10, 20, 80)

结果：

  • 小齿轮：d1=200 mm，da1=220 mm，df1=175 mm
  • 大齿轮：d2=800 mm，da2=820 mm，df2=775 mm
  • 标准中心距 a = 500 mm
  • 传动比 i = 4

场景 4：校验根切风险

需求：设计某传动，初选 m=2，z=14，需判断是否根切。

输入：m=2，z=14

调用：Gear.calcAllParameters(2, 14)

结果：返回对象中 hasUndercutRisk = true，提示 z=14 < 17，标准齿轮会发生根切。建议：① 增大齿数至 ≥17；② 采用正变位齿轮（本工具不支持，需查 GB/T 10095 或相关手册）；③ 改用其他压力角。

场景 5：已知中心距反推齿数和

需求：某箱体中心距已确定为 a=120 mm，选用 m=3，求 z1+z2。

计算：由 a = m·(z1+z2)/2 得 z1+z2 = 2a/m = 2×120/3 = 80。若取传动比 i=3，则 z1=20，z2=60。

验证：Gear.calcCenterDistance(3, 20, 60) 返回 120，符合。

六、常见错误与排查

错误现象可能原因排查方法
抛出 "参数 m 必须为有限数值"传入了 null / undefined / NaN / 字符串 / 对象检查输入类型，确保 m 为 number 类型
抛出 "模数 m 必须 > 0"传入了 0 或负数模数必须为正数
抛出 "模数 m 超出合理范围"m < 0.05 或 m > 100参考 GB/T 1357 标准模数系列
抛出 "齿数 z 必须为整数"传入了小数如 20.5齿数必须为整数
抛出 "齿数 z 必须 >= 1"传入了 0 或负数齿数至少为 1
df 计算结果为负z < 3（如 z=1 或 z=2）z<3 时齿根圆直径公式给出负值，无物理意义。实际工程中 z 至少为 17（避免根切）
结果与手册略有差异浮点精度问题本工具使用 IEEE 754 双精度浮点，结果保留足够有效位。如需精确到 0.001 mm，可对结果做 toFixed(3) 处理

七、数据来源与标准依据

  • 标准齿条齿廓参数（α=20°、ha*=1、c*=0.25）：GB/T 1356-2001 / ISO 53:1998
  • 标准模数系列：GB/T 1357-2008
  • 根切最小齿数 z_min = 2·ha*/sin²α = 2×1/sin²(20°) ≈ 17.097：机械原理教材通用结论
  • 所有几何公式均为标准齿轮（x=0）下的解析式，来源见上述国标及《机械原理》《机械设计》教材
注：本工具未引入任何经验系数或修正值，所有结果均为理论解析值。实际加工还需考虑刀具参数、加工误差、热变形等因素，需查相关工艺手册。

八、局限性说明（本工具不能做什么）

  1. 不支持变位齿轮：所有公式基于变位系数 x=0。若实际采用正/负变位，本工具结果不适用。
  2. 不支持斜齿轮：本工具仅适用于直齿圆柱齿轮。斜齿轮需引入螺旋角 β，法向模数 mn 与端面模数 mt 不同，需另行计算。
  3. 不支持锥齿轮、蜗轮蜗杆、非圆齿轮：这些齿轮类型的几何计算完全不同。
  4. 不计算强度：本工具仅输出几何尺寸，不计算齿面接触强度、齿根弯曲强度（需按 GB/T 3480 / ISO 6336 另行校核）。
  5. 不计算精度等级：不输出公差、偏差（需按 GB/T 10095）。
  6. 不计算重合度、滑动率、干涉：这些需更复杂的几何分析。
  7. 不验证模数是否为 GB/T 1357 标准值：工具允许任意正数模数，但推荐优先采用标准模数系列。
  8. 不处理非标压力角：本工具固定 α=20°。若采用 α=14.5°、α=25° 等，结果不适用。
  9. 浮点精度限制：JavaScript 使用 IEEE 754 双精度浮点，极小/极大数值可能出现精度损失，但对常规工程计算（m∈[0.5, 50]，z∈[17, 500]）影响可忽略。
  10. 不输出图形：本工具仅输出数值，不绘制齿轮轮廓。如需绘图，请结合 SVG/Canvas 另行实现。
