/**
 * 防雷保护范围计算工具（滚球法）
 * 依据：GB 50057-2010《建筑物防雷设计规范》附录 D
 * 版本：v2.1.0-加计算界面 20261022
 */

'use strict';

/**
 * 根据防雷类别获取滚球半径 hr（单位：米）
 * 依据 GB 50057-2010 表 3.0.3 及附录 D.0.1
 *
 * @param {number} category - 防雷类别，取值 1 / 2 / 3
 * @returns {number} 滚球半径 hr（米）
 * @throws {Error} 当 category 不是 1/2/3 时抛出异常
 *
 * @example
 *   getRollingSphereRadius(1); // 30
 *   getRollingSphereRadius(2); // 45
 *   getRollingSphereRadius(3); // 60
 */
function getRollingSphereRadius(category) {
  const map = { 1: 30, 2: 45, 3: 60 };
  if (![1, 2, 3].includes(category)) {
    throw new Error('防雷类别必须为 1、2 或 3，当前值：' + category);
  }
  return map[category];
}

/**
 * 校验数值输入是否合法
 *
 * @param {number} value - 待校验的数值
 * @param {string} name - 参数名称（用于错误提示）
 * @param {Object} [options] - 校验选项
 * @param {number} [options.min] - 最小值（含）
 * @param {number} [options.max] - 最大值（含）
 * @param {boolean} [options.allowZero=true] - 是否允许 0
 * @returns {{ valid: boolean, error: string|null }} 校验结果
 *
 * @example
 *   validateInput(10, 'h', { min: 0 }); // { valid: true, error: null }
 *   validateInput(-1, 'h', { min: 0 }); // { valid: false, error: '...' }
 */
function validateInput(value, name, options) {
  options = options || {};
  const allowZero = options.allowZero !== false;
  if (typeof value !== 'number' || Number.isNaN(value)) {
    return { valid: false, error: name + ' 必须为有效数字' };
  }
  if (!allowZero && value === 0) {
    return { valid: false, error: name + ' 不能为 0' };
  }
  if (options.min !== undefined && value < options.min) {
    return { valid: false, error: name + ' 不能小于 ' + options.min };
  }
  if (options.max !== undefined && value > options.max) {
    return { valid: false, error: name + ' 不能大于 ' + options.max };
  }
  return { valid: true, error: null };
}

/**
 * 计算单支避雷针在地面上的保护半径 r0
 * 公式：r0 = √(h × (2hr − h))，依据 GB 50057-2010 附录 D.0.1
 *
 * @param {number} h - 避雷针高度（米），必须 > 0
 * @param {number} hr - 滚球半径（米），必须 > 0
 * @returns {{ r0: number, effectiveH: number, warning: string|null }}
 *   r0: 地面保护半径（米）
 *   effectiveH: 实际参与计算的高度（当 h>hr 时取 hr）
 *   warning: 警告信息（如 h>hr 时）
 *
 * @example
 *   calcSingleRodGroundRadius(20, 30);
 *   // { r0: 28.284..., effectiveH: 20, warning: null }
 */
function calcSingleRodGroundRadius(h, hr) {
  const v1 = validateInput(h, 'h', { min: 0, allowZero: false });
  if (!v1.valid) throw new Error(v1.error);
  const v2 = validateInput(hr, 'hr', { min: 0, allowZero: false });
  if (!v2.valid) throw new Error(v2.error);

  let effectiveH = h;
  let warning = null;
  if (h > hr) {
    effectiveH = hr;
    warning = '避雷针高度 h 超过滚球半径 hr，按 h = hr 计算（GB 50057-2010 D.0.1）';
  }
  const r0 = Math.sqrt(effectiveH * (2 * hr - effectiveH));
  return { r0: r0, effectiveH: effectiveH, warning: warning };
}

/**
 * 计算单支避雷针在 hx 高度平面上的保护半径 rx
 * 公式：rx = √(h(2hr−h)) − √(hx(2hr−hx))，依据 GB 50057-2010 附录 D.0.1
 *
 * @param {number} h - 避雷针高度（米），必须 > 0
 * @param {number} hr - 滚球半径（米），必须 > 0
 * @param {number} hx - 被保护物高度（米），必须 ≥ 0
 * @returns {{ rx: number, r0: number, effectiveH: number, warning: string|null }}
 *   rx: hx 高度平面上的保护半径（米）
 *   r0: 地面保护半径（米）
 *   effectiveH: 实际参与计算的高度
 *   warning: 警告信息
 *
 * @example
 *   calcSingleRodProtection(20, 30, 10);
 *   // { rx: 16.329..., r0: 28.284..., effectiveH: 20, warning: null }
 */
function calcSingleRodProtection(h, hr, hx) {
  const v1 = validateInput(h, 'h', { min: 0, allowZero: false });
  if (!v1.valid) throw new Error(v1.error);
  const v2 = validateInput(hr, 'hr', { min: 0, allowZero: false });
  if (!v2.valid) throw new Error(v2.error);
  const v3 = validateInput(hx, 'hx', { min: 0 });
  if (!v3.valid) throw new Error(v3.error);

  let effectiveH = h;
  let warning = null;
  if (h > hr) {
    effectiveH = hr;
    warning = '避雷针高度 h 超过滚球半径 hr，按 h = hr 计算';
  }
  if (hx >= effectiveH) {
    return { rx: 0, r0: Math.sqrt(effectiveH * (2 * hr - effectiveH)),
             effectiveH: effectiveH, warning: warning || 'hx ≥ h，保护半径为 0' };
  }
  const r0 = Math.sqrt(effectiveH * (2 * hr - effectiveH));
  const rx = r0 - Math.sqrt(hx * (2 * hr - hx));
  return { rx: rx, r0: r0, effectiveH: effectiveH, warning: warning };
}

/**
 * 计算两支等高避雷针之间保护范围最低点高度 h0
 * 公式：h0 = hr − √((hr−h)² + (D/2)²)，依据 GB 50057-2010 附录 D.0.2
 * 当 h0 < 0 时，表示两针间无法形成联合保护范围
 *
 * @param {number} h - 避雷针高度（米），必须 > 0
 * @param {number} hr - 滚球半径（米），必须 > 0
 * @param {number} D - 两针间距（米），必须 > 0
 * @returns {{ h0: number, canProtect: boolean, maxD: number, warning: string|null }}
 *   h0: 两针间保护范围最低点高度（米），负值表示无联合保护
 *   canProtect: 是否能形成联合保护
 *   maxD: 能形成联合保护的最大间距（米）
 *   warning: 警告信息
 *
 * @example
 *   calcDoubleRodMinHeight(20, 30, 20);
 *   // { h0: 15.857..., canProtect: true, maxD: 56.568..., warning: null }
 */
function calcDoubleRodMinHeight(h, hr, D) {
  const v1 = validateInput(h, 'h', { min: 0, allowZero: false });
  if (!v1.valid) throw new Error(v1.error);
  const v2 = validateInput(hr, 'hr', { min: 0, allowZero: false });
  if (!v2.valid) throw new Error(v2.error);
  const v3 = validateInput(D, 'D', { min: 0, allowZero: false });
  if (!v3.valid) throw new Error(v3.error);

  const effectiveH = Math.min(h, hr);
  const maxD = 2 * Math.sqrt(effectiveH * (2 * hr - effectiveH));
  let warning = null;
  if (h > hr) {
    warning = '避雷针高度 h 超过滚球半径 hr，按 h = hr 计算';
  }

  const discriminant = Math.pow(hr - effectiveH, 2) + Math.pow(D / 2, 2);
  const h0 = hr - Math.sqrt(discriminant);
  const canProtect = h0 >= 0;

  return {
    h0: h0,
    canProtect: canProtect,
    maxD: maxD,
    warning: warning
  };
}

/**
 * 综合计算：给定防雷类别、避雷针高度、被保护物高度，返回完整保护范围信息
 *
 * @param {Object} params - 输入参数
 * @param {number} params.category - 防雷类别（1/2/3）
 * @param {number} params.h - 避雷针高度（米）
 * @param {number} params.hx - 被保护物高度（米）
 * @param {number} [params.D] - 双支避雷针间距（可选，传入则计算双支）
 * @returns {Object} 完整计算结果
 *
 * @example
 *   calcProtection({ category: 2, h: 30, hx: 15 });
 */
function calcProtection(params) {
  if (!params || typeof params !== 'object') {
    throw new Error('参数必须为对象');
  }
  const hr = getRollingSphereRadius(params.category);
  const single = calcSingleRodProtection(params.h, hr, params.hx);
  const result = {
    category: params.category,
    hr: hr,
    h: params.h,
    hx: params.hx,
    single: single
  };
  if (params.D !== undefined) {
    result.double = calcDoubleRodMinHeight(params.h, hr, params.D);
    result.double.D = params.D;
  }
  return result;
}

防雷保护范围计算工具 · 使用说明书

一、工具用途

本工具用于计算避雷针（接闪杆）及避雷带（接闪带）的保护范围，采用 GB 50057-2010《建筑物防雷设计规范》规定的"滚球法"。适用于建筑电气设计、防雷工程审核、既有建筑防雷改造评估等场景，帮助工程师快速判断避雷针能否覆盖指定高度的被保护物，或反推所需避雷针高度。

滚球法的基本原理是：设想一个半径为 hr 的球体沿避雷针滚动，球体不能触及的空间即为保护范围。不同防雷类别对应不同滚球半径：一类 30m、二类 45m、三类 60m。

二、参数详解

参数含义单位取值范围说明

hr滚球半径米30 / 45 / 60由 category 决定，一类 30m、二类 45m、三类 60m
hx被保护物高度米≥ 0需要被保护的设备或区域顶端距地面的高度

rxhx 高度保护半径米—避雷针在 hx 高度平面上的保护范围半径

三、使用场景

场景 1：判断现有避雷针能否保护某设备

某二类防雷建筑屋顶有一支 25m 高的避雷针，下方 10m 高度处有一台精密设备，问是否在保护范围内？

调用 calcSingleRodProtection(25, 45, 10)，返回 rx ≈ 21.65m。若设备距避雷针水平距离小于 21.65m，则在保护范围内。

场景 2：反推所需避雷针高度

某三类防雷建筑需保护 15m 高度处的设备，设备距避雷针水平距离 30m，问避雷针至少多高？

需解方程 rx = 30，即 √(h(120−h)) − √(15×90) = 30。通过迭代或作图可得 h ≈ 38m。本工具不直接提供反算功能，需结合计算结果人工判断。

场景 3：双支避雷针联合保护评估

两支 20m 高的避雷针，间距 30m，一类防雷（hr=30m），问两针间能否形成联合保护？

调用 calcDoubleRodMinHeight(20, 30, 30)，返回 h0 ≈ 5.86m，canProtect = true。说明两针间保护范围最低点距地面 5.86m，低于此高度的设备可被联合保护。

场景 4：防雷改造方案比选

某既有建筑原为三类防雷，现改为二类，原 35m 避雷针是否仍满足要求？

分别计算 category=3 和 category=2 下的 rx，对比保护范围变化。二类 hr=45m 比三类 hr=60m 小，保护范围会缩小，需重新评估。

场景 5：避雷带（接闪带）保护范围估算

避雷带的保护范围可近似按单支避雷针计算，将避雷带视为一系列连续的避雷针。本工具的单支避雷针计算结果可作为避雷带保护范围的参考上限（实际保护范围可能略小，需查标准原文确认）。

四、常见错误

  • 防雷类别选错：类别直接决定滚球半径，选错会导致保护范围计算完全错误。应依据 GB 50057-2010 第 3 章严格判定，不可凭经验估计。
  • 忽略 h > hr 的截断：当避雷针高度超过滚球半径时，保护范围不再随高度增加而增大，工具已自动按 h = hr 计算，但用户需注意返回的 warning 提示。
  • 混淆 hx 与 h：hx 是被保护物高度，h 是避雷针高度。当 hx ≥ h 时，保护半径 rx = 0，这是正常的物理现象。
  • 双针间距过大：当 D > 2r0 时，两针间无法形成联合保护，工具返回 canProtect = false，此时应减小间距或增加避雷针数量。
  • 单位错误：所有输入输出单位均为"米"，不可混用厘米、毫米等其他单位。

五、数据依据

本工具所有算法公式均来源于 GB 50057-2010《建筑物防雷设计规范》：

  • 滚球半径取值：附录 D.0.1，一类 30m、二类 45m、三类 60m
  • 单支避雷针保护范围：附录 D.0.1，公式 r0 = √(h(2hr−h))，rx = √(h(2hr−h)) − √(hx(2hr−hx))
  • 双支等高避雷针：附录 D.0.2，两针间保护范围最低点高度 h0 = hr − √((hr−h)² + (D/2)²)
注：部分细节公式（如双支不等高避雷针、多支避雷针、避雷线等）本工具暂未实现，需查标准原文确认。

六、局限性

  • 仅实现滚球法基本公式：未涵盖 GB 50057-2010 中所有接闪器布置形式（如网格形接闪带、架空接闪线等），复杂布置需查标准原文或采用专业软件。
  • 不考虑地形影响：实际工程中，山坡、峡谷、水域等地形会显著影响雷电先导发展路径，本工具未考虑这些因素。
  • 不考虑相邻建筑影响：多栋建筑相邻时，雷电可能被相邻较高建筑吸引，本工具仅计算单栋建筑独立情况。
  • 不考虑雷电参数的随机性：雷电流幅值、极性、波形等具有强随机性，滚球法本身是一种几何简化模型，不能反映所有雷电物理过程。
  • 不适用于特殊结构：超高层建筑（高度超过 100m）、高耸构筑物（如烟囱、塔架）、易燃易爆场所等，需按专门规范或专项评估执行。

七、精度与误差说明（有限精准原则）

现实变量是无限的，本工具只做到"方法准、输入项准、结果给误差范围"，不承诺绝对精确。以下逐项说明：

① 算准了什么：

  • GB 50057-2010 附录 D 给出的滚球法几何公式，工具严格按公式实现，公式层面的计算误差仅来自 IEEE 754 双精度浮点运算（约 15–17 位有效数字），对于工程应用（米级精度）可视为无误差。
  • 防雷类别与滚球半径的映射关系（1→30m、2→45m、3→60m）严格按标准执行。

② 哪些因素没算进去：

  • 雷电先导的实际发展路径受大气电场、湿度、气压、地面导电率等多种因素影响，滚球法仅是一种几何等效模型，不能反映真实物理过程。
  • 避雷针实际保护范围受针尖形状、材质、腐蚀程度、安装垂直度等施工因素影响，工具假设理想针尖。
  • 未考虑雷电侧击、绕击等小概率事件。
  • 未考虑接闪器老化、维护状态对保护效果的影响。

③ 误差大概多大：

  • 公式计算误差：小于 10⁻¹² 米（双精度浮点极限），可忽略。
  • 模型误差：滚球法本身是一种工程近似方法，国内外研究表明其与实际保护范围的偏差可达 ±10%–20%，在边缘区域（保护范围边界附近）偏差可能更大。
  • 工程建议：重要设施应在计算结果基础上留有一定裕量（如将保护半径乘以 0.8–0.9 的安全系数），具体系数需根据工程经验或专项评估确定。
重要提示：本工具输出结果仅供工程参考，不可替代专业防雷设计审查。涉及人身安全、重要设施的防雷工程，必须由具备资质的防雷专业机构进行设计、审核和验收。
