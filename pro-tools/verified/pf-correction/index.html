/**
 * 功率因数补偿计算器
 * 核心公式：Qc = P × (tanφ₁ − tanφ₂)
 * 参考：GB/T 15576-2020《低压无功功率动态补偿装置》
 * 注意：本工具为工程估算，不承诺绝对精确（详见说明书《精度与误差说明》）
 */

/**
 * 将功率因数 cosφ 转换为 tanφ
 * 数学关系：tanφ = √(1 − cos²φ) / cosφ
 *
 * @param {number} cosPhi - 功率因数，取值范围 (0, 1]
 * @returns {number} tanφ 的值；当 cosφ = 1 时返回 0
 * @throws {Error} 当 cosPhi 不在 (0, 1] 范围内时抛出
 *
 * @example
 * cosToTan(0.8); // ≈ 0.75
 * cosToTan(1.0); // 0
 */
function cosToTan(cosPhi) {
  if (typeof cosPhi !== 'number' || Number.isNaN(cosPhi)) {
    throw new Error('cosPhi 必须是数字');
  }
  if (cosPhi <= 0 || cosPhi > 1) {
    throw new Error('cosPhi 必须在 (0, 1] 范围内');
  }
  if (cosPhi === 1) return 0;
  const sinPhi = Math.sqrt(1 - cosPhi * cosPhi);
  return sinPhi / cosPhi;
}

/**
 * 校验补偿计算的输入参数
 *
 * @param {number} P - 有功功率，单位 kW，必须 > 0
 * @param {number} cosPhi1 - 当前功率因数，(0, 1]
 * @param {number} cosPhi2 - 目标功率因数，(0, 1]
 * @returns {{ ok: boolean, message: string }} 校验结果
 */
function validateInputs(P, cosPhi1, cosPhi2) {
  if ([P, cosPhi1, cosPhi2].some(v => typeof v !== 'number' || Number.isNaN(v))) {
    return { ok: false, message: '所有输入必须是数字' };
  }
  if (P <= 0) return { ok: false, message: '有功功率 P 必须大于 0' };
  if (cosPhi1 <= 0 || cosPhi1 > 1) return { ok: false, message: 'cosφ₁ 必须在 (0, 1]' };
  if (cosPhi2 <= 0 || cosPhi2 > 1) return { ok: false, message: 'cosφ₂ 必须在 (0, 1]' };
  if (cosPhi2 < cosPhi1) {
    return { ok: false, message: '目标功率因数不应低于当前值（否则为反向补偿）' };
  }
  return { ok: true, message: 'OK' };
}

/**
 * 计算所需无功补偿容量 Qc
 * 公式：Qc = P × (tanφ₁ − tanφ₂)
 *
 * @param {number} P - 有功功率，单位 kW
 * @param {number} cosPhi1 - 当前功率因数，(0, 1]
 * @param {number} cosPhi2 - 目标功率因数，(0, 1]，必须 ≥ cosPhi1
 * @returns {{ Qc_kvar: number, tanPhi1: number, tanPhi2: number }}
 *          Qc_kvar 为所需补偿容量（kvar）
 * @throws {Error} 输入非法时抛出
 *
 * @example
 * calculateCompensation(100, 0.75, 0.95);
 * // { Qc_kvar: 52.56..., tanPhi1: 0.8819..., tanPhi2: 0.3286... }
 */
function calculateCompensation(P, cosPhi1, cosPhi2) {
  const v = validateInputs(P, cosPhi1, cosPhi2);
  if (!v.ok) throw new Error(v.message);
  const tanPhi1 = cosToTan(cosPhi1);
  const tanPhi2 = cosToTan(cosPhi2);
  const Qc_kvar = P * (tanPhi1 - tanPhi2);
  return { Qc_kvar, tanPhi1, tanPhi2 };
}

/**
 * 由补偿容量计算电容器容量（电容值，单位 μF）
 *
 * 单相：C = Qc / (2π × f × U²)  （U 为相电压，V）
 * 三相（星形接法，线电压 U）：C = Qc / (3 × 2π × f × (U/√3)²)
 *                              = Qc / (2π × f × U²)  （化简后与单相形式相同，但 U 为线电压）
 * 三相（三角形接法，线电压 U）：C = Qc / (3 × 2π × f × U²)
 *
 * @param {number} Qc_kvar - 补偿容量，单位 kvar
 * @param {number} voltage - 电压，单位 V（单相为相电压；三相为线电压）
 * @param {number} [frequency=50] - 频率，单位 Hz，默认 50
 * @param {'single'|'three-star'|'three-delta'} [connection='three-star'] - 接线方式
 * @returns {number} 电容值，单位 μF
 * @throws {Error} 输入非法时抛出
 */
function calculateCapacitance(Qc_kvar, voltage, frequency = 50, connection = 'three-star') {
  if (Qc_kvar < 0) throw new Error('Qc 不能为负');
  if (voltage <= 0) throw new Error('电压必须大于 0');
  if (frequency <= 0) throw new Error('频率必须大于 0');
  const Qc_var = Qc_kvar * 1000;
  const omega = 2 * Math.PI * frequency;
  let C_farad;
  if (connection === 'single') {
    C_farad = Qc_var / (omega * voltage * voltage);
  } else if (connection === 'three-star') {
    C_farad = Qc_var / (omega * voltage * voltage);
  } else if (connection === 'three-delta') {
    C_farad = Qc_var / (3 * omega * voltage * voltage);
  } else {
    throw new Error('接线方式参数错误');
  }
  return C_farad * 1e6; // 转为 μF
}

/**
 * 一站式计算：输入 P、cosφ₁、cosφ₂、电压等，返回完整结果
 *
 * @param {Object} params
 * @param {number} params.P - 有功功率 kW
 * @param {number} params.cosPhi1 - 当前功率因数
 * @param {number} params.cosPhi2 - 目标功率因数
 * @param {number} [params.voltage=400] - 电压 V（三相线电压默认 400V）
 * @param {number} [params.frequency=50] - 频率 Hz
 * @param {string} [params.connection='three-star'] - 接线方式
 * @returns {Object} 包含 Qc_kvar、电容值、校验信息等的完整结果
 */
function computeFull(params) {
  const { P, cosPhi1, cosPhi2, voltage = 400, frequency = 50, connection = 'three-star' } = params;
  const v = validateInputs(P, cosPhi1, cosPhi2);
  if (!v.ok) return { ok: false, error: v.message };
  const { Qc_kvar, tanPhi1, tanPhi2 } = calculateCompensation(P, cosPhi1, cosPhi2);
  const capacitance_uF = calculateCapacitance(Qc_kvar, voltage, frequency, connection);
  const S1 = P / cosPhi1;
  const S2 = P / cosPhi2;
  return {
    ok: true,
    Qc_kvar, tanPhi1, tanPhi2,
    capacitance_uF,
    S1_kVA: S1, S2_kVA: S2,
    S_reduced_kVA: S1 - S2,
    voltage, frequency, connection
  };
}

// 兼容浏览器与 Node.js
if (typeof module !== 'undefined') {
  module.exports = { cosToTan, validateInputs, calculateCompensation, calculateCapacitance, computeFull };
}

第三部分：使用说明书

一、本工具干什么用

本工具用于估算低压交流系统中，将功率因数从当前值 cosφ₁ 提升到目标值 cosφ₂ 所需的无功补偿容量 Qc（单位 kvar），以及在给定电压、频率、接线方式下对应的电容器容量（单位 μF）。

典型用途包括：

  • 工厂配电房初步估算补偿柜容量
  • 电气设计阶段方案比选
  • 教学演示功率因数补偿原理
  • 现场工程师快速复核已有方案
重要前提：本工具基于理想正弦波、稳态、三相对称假设，结果仅供工程估算参考，不能直接作为施工图或采购依据。正式工程须由具备资质的设计单位按相关标准核算。

二、参数详解

三、核心公式

补偿容量：

Qc = P × (tanφ₁ − tanφ₂)

其中 tanφ = √(1 − cos²φ) / cosφ，可由 cosφ 直接换算。

电容器容量（三相星形接法，线电压 U）：

C = Qc / (2π × f × U²)（单位 F，工程常用 μF）

三角形接法时，每相电容为星形的 1/3。

四、5 个典型使用场景

场景 1：小型机加工车间

某车间总装机有功约 80 kW，实测平均功率因数 0.72，供电局要求 ≥ 0.90。代入：

  • tanφ₁ = cosToTan(0.72) ≈ 0.9638
  • tanφ₂ = cosToTan(0.90) ≈ 0.4843
  • Qc = 80 × (0.9638 − 0.4843) ≈ 38.4 kvar

工程上可选 40 kvar 补偿柜（含若干步进电容）。注意：实际选型需考虑谐波、负载波动，容量通常留 10%～20% 裕度。

场景 2：商业写字楼配电

变压器容量 800 kVA，负载有功 500 kW，当前 cosφ = 0.82，目标 0.95。

  • Qc = 500 × (tan(arccos 0.82) − tan(arccos 0.95)) ≈ 500 × (0.6993 − 0.3287) ≈ 185 kvar

建议采用自动投切的无功补偿柜，按 6～12 路步进配置。

场景 3：居民小区变压器低压侧

负荷波动大，白天 cosφ 低、夜间高。建议按最大负荷计算，并采用动态补偿（接触器或晶闸管投切）。本工具按最大负荷 P 代入即可。

场景 4：光伏并网点补偿

光伏逆变器本身可发无功，但容量有限。当电网要求并网点 cosφ ≥ 0.95 时，需核算逆变器剩余无功能力，不足部分由 SVG 或电容器补足。本工具可估算缺口，但需结合逆变器曲线。

场景 5：电焊机集中使用场景

电焊机 cosφ 可低至 0.4～0.6，且冲击性强。按最大同时使用台数估算 P，目标 cosφ 取 0.9 左右。注意：此类负载谐波大，须配套电抗器，否则电容器易损坏。

七、局限性（老实写）

本工具没有考虑以下因素，实际工程必须另行核算：

  • 谐波电流对电容器的影响（可能引发谐振）
  • 三相不平衡导致的分相补偿需求
  • 负载随时间波动的动态特性
  • 电容器老化、温度漂移导致的容量衰减
  • 线路阻抗、变压器损耗
  • 电压波动对补偿效果的影响
  • 过补偿风险（轻载时 cosφ 可能超前）
  • 电容器投切涌流对开关设备的影响
  • 与 SVG、同步调相机等动态设备的配合

因此，本工具结果只能作为方案初期的数量级估算，不能替代详细设计。

八、精度与误差说明（必读）

现实变量是无限的，本工具只做到「方法准、输入项准、结果给误差范围」，不承诺绝对精确。

① 算准了什么

  • 数学公式本身：Qc = P × (tanφ₁ − tanφ₂) 在理想正弦、稳态、三相对称条件下严格成立
  • 浮点运算精度：JavaScript 双精度浮点，相对误差约 10⁻¹⁵ 量级
  • 输入参数若准确，则输出在数学意义上准确

② 哪些因素没算进去

  • 实际负载的 cosφ 是波动的，本工具按单点值计算
  • 电容器实际容量与标称值有偏差（通常 ±5%～±10%）
  • 电网电压波动导致实际补偿容量偏离设计值（Q ∝ U²）
  • 谐波会使电容器实际电流增大，发热加速老化
  • 温度影响电容器介质，容量随温度变化
  • 三相不平衡时，总补偿容量不等于各相之和的简单关系

③ 误差大概多大

来源典型误差范围
数学计算本身< 10⁻¹⁰（可忽略）
输入 cosφ 测量误差 ±0.01Qc 误差约 2%～5%
输入 P 测量误差 ±5%Qc 误差 ±5%
电容器标称偏差±5%～±10%
电压波动 ±5%实际补偿容量波动约 ±10%
综合工程误差通常 ±10%～±20%

因此，工程选型时通常在设计值基础上留 10%～20% 裕度，并采用多路步进投切以适应负载变化。

再次强调：本工具不承诺「精确计算」「绝对准确」「100% 无误」。所有结果均为工程估算，正式工程须由具备资质的设计单位按现行标准核算，并经供电部门审核。
